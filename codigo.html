Firmware da Caixinha do Tchauzinho, organizado e com realce de sintaxe.

MarS_Tchauzinho.ino Arduino C++ 
#include <Servo.h>

Servo servo_base;
Servo servo_braco;

int i;

// Pinos do sensor ultrassônico
int pinTrig = 7;
int pinEcho = 8;

// Pinos dos servos
int pinServo_base = 2;
int pinServo_braco = 3;

// Posições dos servos
int posOff_base = 0;
int posOff_braco = 0;
int posB_e1 = 90;

// Distância medida pelo sensor
long duracao;
int distancia;

// Estado da detecção
bool pessoaDetectada = false;

//---------------------------------
// Medir distância
//---------------------------------
int medir_distancia(){
  digitalWrite(pinTrig, LOW);
  delayMicroseconds(2);

  digitalWrite(pinTrig, HIGH);
  delayMicroseconds(10);

  digitalWrite(pinTrig, LOW);

  duracao = pulseIn(pinEcho, HIGH, 30000);

  if(duracao == 0){
    return 0;
  }

  distancia = duracao * 0.034 / 2;
  return distancia;
}

//---------------------------------
// Sequência do tchauzinho
//---------------------------------
void ativar_braco(){
  // Levanta o braço
  for(i = posOff_braco; i <= 90; i++){
    servo_braco.write(i);
    delay(5);
  }

  // Primeiro aceno
  for(i = 90; i >= 0; i--){
    servo_braco.write(i);
    delay(10);
  }
  for(i = 0; i <= 90; i++){
    servo_braco.write(i);
    delay(10);
  }

  // Segundo aceno
  for(i = 90; i >= 0; i--){
    servo_braco.write(i);
    delay(10);
  }
  for(i = 0; i <= 90; i++){
    servo_braco.write(i);
    delay(10);
  }

  // Abaixa o braço
  for(i = 90; i >= posOff_braco; i--){
    servo_braco.write(i);
    delay(5);
  }
}

//---------------------------------
// SETUP
//---------------------------------
void setup(){
  servo_base.attach(pinServo_base);
  servo_braco.attach(pinServo_braco);

  pinMode(pinTrig, OUTPUT);
  pinMode(pinEcho, INPUT);

  servo_base.write(posOff_base);
  servo_braco.write(posOff_braco);
}

//---------------------------------
// LOOP PRINCIPAL
//---------------------------------
void loop(){
  distancia = medir_distancia();

  // Pessoa detectada a até 15 cm
  if(distancia > 0 && distancia <= 15 && pessoaDetectada == false){
    pessoaDetectada = true;

    // Posiciona a base e dá tchau
    servo_base.write(posB_e1);
    ativar_braco();
  }

  // Faixa intermediária: mantém o estado atual
  else if(distancia > 15 && distancia < 25){
    // Não faz absolutamente nada.
  }

  // Pessoa afastada: rearma o sistema
  else if(distancia >= 25 && pessoaDetectada == true){
    pessoaDetectada = false;

    servo_base.write(posOff_base);
    servo_braco.write(posOff_braco);
  }
}
DETECÇÃO
1–15 cm

Inicia o gesto uma vez.

MOVIMENTO
2 acenos

Braço entre 0° e 90°.

REARME
≥25 cm

Base e braço retornam a 0°.
